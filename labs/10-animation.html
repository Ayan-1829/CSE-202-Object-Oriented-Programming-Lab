Animation with Threads

Animate bouncing balls: a worker thread updates positions about 60 times a second, a custom panel draws each frame with double buffering, and Start, Pause and Reset buttons control it.

Before we start

Why this lab

A film is just still pictures shown quickly. How can a Java program make a ball look like it is moving?

Draw it, move it a little, draw it again — about 60 times a second. A worker thread runs that loop while Swing keeps the window responsive.

Theory 1 of 5

Frame-based animation

  • Each frame: update the state, redraw, wait. Repeated quickly, the eye sees smooth motion.
  • About 16 ms per frame gives roughly 60 frames per second.

Frame-based animation

x = x + vx;
y = y + vy;

Theory 2 of 5

The worker thread and the EDT

while (true) {
    panel.updateAnimation();   // move the balls
    panel.repaint();           // ask Swing to draw
    Thread.sleep(16);          // ~60 FPS
}
  • The loop runs in AnimationThread so the Event Dispatch Thread stays free for clicks and painting.
  • Both threads touch the balls, so AnimationPanel’s methods are synchronized.

Double buffering

BufferedImage buffer = new BufferedImage(w, h, BufferedImage.TYPE_INT_RGB);
Graphics2D g2d = buffer.createGraphics();

  • Swing already double-buffers; doing it by hand shows the idea.

Applets and modern animation

Implementation design

Program design

Implementation design

Execution flow

  1. main creates AnimationApp on the EDT
  2. AnimationApp builds the panel and buttons and starts a daemon AnimationThread
  3. The thread loops: updateAnimation(), repaint(), sleep the rest of the 16 ms frame
  4. updateAnimation moves balls only while running and handles collisions
  5. Start / Pause flip the running flag; Reset restores the three balls
  6. Closing the window ends the program (the thread is a daemon)

Ball = model, AnimationPanel = rendering + state, AnimationThread = loop, AnimationApp = controls.

Ball-to-ball collisions

public boolean collidesWith(Ball other) {
    double dx = this.x - other.x;
    double dy = this.y - other.y;
    double distance = Math.sqrt(dx * dx + dy * dy);
    return distance <= this.radius + other.radius;
}

public void bounceOff(Ball other) {
    double tempVx = this.vx;
    double tempVy = this.vy;
    this.vx = other.vx;
    this.vy = other.vy;
    other.vx = tempVx;
    other.vy = tempVy;
}
  • Touching when the centres are closer than the radii added together.
  • Swapping velocities is a simple bounce.

import java.awt.BasicStroke;
import java.awt.Color;
import java.awt.Graphics2D;

public class Ball {

    private double x;
    private double y;
    private double vx;
    private double vy;
    private int radius;
    private Color colour;

    public Ball(double x, double y, double vx, double vy, int radius, Color colour) {
        this.x = x;
        this.y = y;
        this.vx = vx;
        this.vy = vy;
        this.radius = (radius > 0) ? radius : 15;
        this.colour = colour;
    }

    public void update(int panelWidth, int panelHeight) {
        x = x + vx;
        y = y + vy;
        if (x - radius < 0 || x + radius > panelWidth) {
            vx = -vx;
            x = Math.max(radius, Math.min(panelWidth - radius, x));
        }
        if (y - radius < 0 || y + radius > panelHeight) {
            vy = -vy;
            y = Math.max(radius, Math.min(panelHeight - radius, y));
        }
    }

    public void draw(Graphics2D g) {
        int diameter = 2 * radius;
        g.setColor(colour);
        g.fillOval((int) (x - radius), (int) (y - radius), diameter, diameter);
        g.setColor(Color.BLACK);
        g.setStroke(new BasicStroke(2));
        g.drawOval((int) (x - radius), (int) (y - radius), diameter, diameter);
    }

    public boolean collidesWith(Ball other) {
        double dx = this.x - other.x;
        double dy = this.y - other.y;
        double distance = Math.sqrt(dx * dx + dy * dy);
        return distance <= this.radius + other.radius;
    }

    public void bounceOff(Ball other) {
        double tempVx = this.vx;
        double tempVy = this.vy;
        this.vx = other.vx;
        this.vy = other.vy;
        other.vx = tempVx;
        other.vy = tempVy;
    }

    public double getX() { return x; }
    public double getY() { return y; }
    public int getRadius() { return radius; }
}
import javax.swing.JPanel;
import java.awt.*;
import java.awt.image.BufferedImage;
import java.util.ArrayList;

public class AnimationPanel extends JPanel {

    private ArrayList<Ball> balls;
    private boolean running;

    public AnimationPanel() {
        balls = new ArrayList<Ball>();
        running = false;
        initializeBalls();
        setBackground(Color.WHITE);
    }

    private void initializeBalls() {
        balls.clear();
        balls.add(new Ball(100, 100, 3.0, 2.0, 15, Color.RED));
        balls.add(new Ball(300, 150, -2.5, 3.5, 15, Color.BLUE));
        balls.add(new Ball(200, 300, 2.0, -2.0, 15, Color.GREEN));
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        if (getWidth() <= 0 || getHeight() <= 0) {
            return;
        }
        BufferedImage buffer = new BufferedImage(getWidth(), getHeight(),
                BufferedImage.TYPE_INT_RGB);
        Graphics2D g2d = buffer.createGraphics();
        g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING,
                RenderingHints.VALUE_ANTIALIAS_ON);
        g2d.setColor(Color.WHITE);
        g2d.fillRect(0, 0, getWidth(), getHeight());
        drawGrid(g2d);

        synchronized (this) {
            for (Ball ball : balls) {
                ball.draw(g2d);
            }
            g2d.setColor(Color.BLACK);
            g2d.drawString("Balls: " + balls.size() + " | "
                    + (running ? "Running" : "Paused"), 10, getHeight() - 10);
        }
        g.drawImage(buffer, 0, 0, null);
        g2d.dispose();
    }

    private void drawGrid(Graphics2D g2d) {
        g2d.setColor(new Color(220, 220, 220));
        for (int x = 0; x < getWidth(); x += 50) {
            g2d.drawLine(x, 0, x, getHeight());
        }
        for (int y = 0; y < getHeight(); y += 50) {
            g2d.drawLine(0, y, getWidth(), y);
        }
    }

    public synchronized void updateAnimation() {
        if (!running) {
            return;
        }
        int width = getWidth();
        int height = getHeight();
        if (width <= 0 || height <= 0) {
            return;
        }
        for (Ball ball : balls) {
            ball.update(width, height);
        }
        for (int i = 0; i < balls.size(); i++) {
            for (int j = i + 1; j < balls.size(); j++) {
                Ball first = balls.get(i);
                Ball second = balls.get(j);
                if (first.collidesWith(second)) {
                    first.bounceOff(second);
                }
            }
        }
    }

    public synchronized void startAnimation() { running = true; }

    public synchronized void pauseAnimation() { running = false; }

    public synchronized void resetAnimation() {
        running = false;
        initializeBalls();
        repaint();
    }

    public synchronized boolean isRunning() { return running; }

    public synchronized int getBallCount() { return balls.size(); }
}
public class AnimationThread implements Runnable {

    private AnimationPanel panel;

    private static final int TARGET_FPS = 60;
    private static final long FRAME_TIME_MS = 1000 / TARGET_FPS;

    public AnimationThread(AnimationPanel panel) {
        this.panel = panel;
    }

    @Override
    public void run() {
        while (true) {
            long frameStartTime = System.currentTimeMillis();

            panel.updateAnimation();
            panel.repaint();

            long elapsedTime = System.currentTimeMillis() - frameStartTime;
            long sleepTime = FRAME_TIME_MS - elapsedTime;
            if (sleepTime > 0) {
                try {
                    Thread.sleep(sleepTime);
                } catch (InterruptedException e) {
                    System.out.println("Animation thread interrupted.");
                    Thread.currentThread().interrupt();
                    break;
                }
            }
        }
    }
}
import javax.swing.*;
import java.awt.*;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;

public class AnimationApp extends JFrame {

    private AnimationPanel animationPanel;
    private JButton startButton;
    private JButton pauseButton;
    private JButton resetButton;
    private JLabel statusLabel;
    private Thread animationThread;

    public AnimationApp() {
        setTitle("Bouncing Balls Animation");
        setSize(800, 600);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);
        setupUI();
        addActionListeners();
        startWorkerThread();
        setVisible(true);
    }

    private void setupUI() {
        JPanel mainPanel = new JPanel(new BorderLayout(10, 10));
        mainPanel.setBorder(BorderFactory.createEmptyBorder(10, 10, 10, 10));

        JPanel topPanel = new JPanel(new FlowLayout(FlowLayout.LEFT));
        JLabel titleLabel = new JLabel("Bouncing Balls Animation");
        titleLabel.setFont(new Font("Arial", Font.BOLD, 16));
        topPanel.add(titleLabel);

        animationPanel = new AnimationPanel();
        animationPanel.setBorder(BorderFactory.createLineBorder(Color.BLACK, 2));

        JPanel controlPanel = new JPanel(new BorderLayout());
        JPanel buttonPanel = new JPanel(new FlowLayout(FlowLayout.LEFT));
        startButton = new JButton("Start");
        pauseButton = new JButton("Pause");
        resetButton = new JButton("Reset");
        buttonPanel.add(startButton);
        buttonPanel.add(pauseButton);
        buttonPanel.add(resetButton);

        statusLabel = new JLabel("Status: Initialized");
        controlPanel.add(buttonPanel, BorderLayout.CENTER);
        controlPanel.add(statusLabel, BorderLayout.SOUTH);

        mainPanel.add(topPanel, BorderLayout.NORTH);
        mainPanel.add(animationPanel, BorderLayout.CENTER);
        mainPanel.add(controlPanel, BorderLayout.SOUTH);
        setContentPane(mainPanel);
    }

    private void addActionListeners() {
        startButton.addActionListener(new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent e) {
                animationPanel.startAnimation();
                updateStatus("Running");
            }
        });
        pauseButton.addActionListener(new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent e) {
                animationPanel.pauseAnimation();
                updateStatus("Paused");
            }
        });
        resetButton.addActionListener(new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent e) {
                animationPanel.resetAnimation();
                updateStatus("Reset");
            }
        });
    }

    private void startWorkerThread() {
        animationThread = new Thread(new AnimationThread(animationPanel));
        animationThread.setDaemon(true);
        animationThread.start();
    }

    private void updateStatus(String status) {
        statusLabel.setText("Status: " + status + " | Balls: " + animationPanel.getBallCount());
    }
}
