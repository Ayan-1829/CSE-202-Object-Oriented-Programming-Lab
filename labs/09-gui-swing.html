Build the game · Chapter 9

Today’s build

  • Three files: Sprites draws every character from shapes, GamePanel is the drawing area and key listener, GameFrame is the window with its label and buttons.
  • Nothing moves on its own yet: the panel repaints only after a key or a click.

What the program does

  1. Build the window on the Event Dispatch Thread
  2. Lay out the score label, the game panel and the buttons
  3. Draw the sky, the city, the ground, three drones and the commando
  4. Move with the arrow keys; Space or the Fire button shoots a laser
  5. The laser stops at the first drone in line and destroys it
  6. Reset puts everything back

@Override
public void keyPressed(KeyEvent e) {
    switch (e.getKeyCode()) {
        case KeyEvent.VK_LEFT:  playerX = Math.max(30, playerX - 10); break;
        case KeyEvent.VK_RIGHT: playerX = Math.min(W - 60, playerX + 10); break;
        case KeyEvent.VK_UP:    playerY = Math.max(120, playerY - 10); break;   // jump jets
        case KeyEvent.VK_DOWN:  playerY = Math.min(GROUND, playerY + 10); break;
        case KeyEvent.VK_SPACE: fire(); return;
        default: return;
    }
    laserY = -1;                        // moving clears the last shot
    repaint();
}

public void keyReleased(KeyEvent e) { }

@Override
public void keyTyped(KeyEvent e) { }

  • keyReleased and keyTyped must exist, even if empty.

public static void drawBackground(Graphics2D g, int w, int h, int ground) {
    g.setPaint(new GradientPaint(0, 0, SKY_TOP, 0, ground, SKY_BOTTOM));
    g.fillRect(0, 0, w, ground);
    g.setColor(CITY);
    int[] tops = { 60, 90, 40, 110, 70, 50, 95, 65 };
    int bw = w / tops.length;
    for (int i = 0; i < tops.length; i++) {
        g.fillRect(i * bw + 6, ground - tops[i], bw - 12, tops[i]);
    }
    g.setColor(DIRT);
    g.fillRect(0, ground, w, h - ground);
    g.setColor(GRASS);
    g.fillRect(0, ground, w, 4);
}
  • Extend JPanel and override paintComponent(Graphics g).
  • Draw with setColor, fillRect, fillOval, drawLine, drawString.
  • (0, 0) is the top-left corner; y grows downward.
  • A GradientPaint fills the sky.

paintComponent(): drawing the scene

@Override
protected void paintComponent(Graphics g0) {
    super.paintComponent(g0);           // clear the old picture first
    Graphics2D g = (Graphics2D) g0;
    g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
    Sprites.drawBackground(g, W, H, GROUND);
    for (Point d : drones) {
        Sprites.drawDrone(g, d.x, d.y);
    }
    Sprites.drawCommando(g, playerX, playerY, laserY >= 0);
    if (laserY >= 0) {
        Sprites.drawLaser(g, playerX + 36, laserY, laserEndX);
    }
    if (drones.isEmpty()) {
        g.setColor(Color.WHITE);
        g.setFont(new Font("SansSerif", Font.BOLD, 32));
        g.drawString("All drones down!", W / 2 - 130, H / 2);
    }
}
  • super.paintComponent clears the old picture.
  • Draw order: background, drones, player, laser.

repaint() and the Event Dispatch Thread

// never call paintComponent yourself
playerX += 10;
repaint();               // "please redraw soon"

public static void main(String[] args) {
    SwingUtilities.invokeLater(() -> new GameFrame().setVisible(true));
}
  • Never call paintComponent yourself: call repaint() and Swing draws soon.
  • Swing work happens on one thread, the Event Dispatch Thread (EDT).
  • Start the GUI with SwingUtilities.invokeLater(…).
  • repaint() asks Swing to call paintComponent on its own thread.

public void fire() {
    laserY = playerY - 36;
    laserEndX = W;
    Point hit = null;
    for (Point d : drones) {
        boolean ahead = d.x > playerX;
        boolean inLine = Math.abs(d.y - laserY) < 16;

}

// The drawing area: a JPanel that paints the arena and listens to the keyboard
public class GamePanel extends JPanel implements KeyListener {
    public static final int W = 760, H = 380, GROUND = 330;

        drones.add(new Point(640, 200));

            score += 100;

            case KeyEvent.VK_LEFT:  playerX = Math.max(30, playerX - 10); break;
            case KeyEvent.VK_RIGHT: playerX = Math.min(W - 60, playerX + 10); break;
            case KeyEvent.VK_UP:    playerY = Math.max(120, playerY - 10); break;   // jump jets
            case KeyEvent.VK_DOWN:  playerY = Math.min(GROUND, playerY + 10); break;
            case KeyEvent.VK_SPACE: fire(); return;
            default: return;
        }
        laserY = -1;                        // moving clears the last shot
        repaint();
    }

    @Override
    public void keyReleased(KeyEvent e) { }

    @Override
    public void keyTyped(KeyEvent e) { }
}

import java.awt.Color;
import java.awt.Graphics;

public abstract class Shape {

    protected int x;
    protected int y;
    protected Color colour;

    public Shape(int x, int y, Color colour) {
        this.x = x;
        this.y = y;
        this.colour = colour;
    }

    public abstract void draw(Graphics g);
}
import java.awt.Color;
import java.awt.Graphics;

public class RectangleShape extends Shape {

    private int width;
    private int height;

    public RectangleShape(int x, int y, int width, int height, Color colour) {
        super(x, y, colour);
        this.width = (width > 0) ? width : 100;
        this.height = (height > 0) ? height : 80;
    }

    @Override
    public void draw(Graphics g) {
        g.setColor(colour);
        g.fillRect(x, y, width, height);
        g.setColor(Color.BLACK);
        g.drawRect(x, y, width, height);
    }
}
import java.awt.Color;
import java.awt.Graphics;

public class CircleShape extends Shape {

    private int radius;

    public CircleShape(int x, int y, int radius, Color colour) {
        super(x, y, colour);
        this.radius = (radius > 0) ? radius : 50;
    }

    @Override
    public void draw(Graphics g) {
        int diameter = 2 * radius;
        g.setColor(colour);
        g.fillOval(x - radius, y - radius, diameter, diameter);
        g.setColor(Color.BLACK);
        g.drawOval(x - radius, y - radius, diameter, diameter);
    }
}
import javax.swing.JPanel;
import java.awt.Color;
import java.awt.Graphics;
import java.util.ArrayList;

public class DrawingPanel extends JPanel {

    private ArrayList<Shape> shapes;
    private Color currentColor;

    public DrawingPanel() {
        shapes = new ArrayList<Shape>();
        currentColor = Color.BLUE;
        setBackground(Color.WHITE);
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        drawGrid(g);
        for (Shape shape : shapes) {
            shape.draw(g);
        }
    }

    private void drawGrid(Graphics g) {
        g.setColor(new Color(220, 220, 220));
        for (int x = 0; x < getWidth(); x += 50) {
            g.drawLine(x, 0, x, getHeight());
        }
        for (int y = 0; y < getHeight(); y += 50) {
            g.drawLine(0, y, getWidth(), y);
        }
    }

    public void addRectangle(int x, int y, int width, int height) {
        shapes.add(new RectangleShape(x, y, width, height, currentColor));
        repaint();
    }

    public void addCircle(int x, int y, int radius) {
        shapes.add(new CircleShape(x, y, radius, currentColor));
        repaint();
    }

    public void clear() {
        shapes.clear();
        repaint();
    }

    public void setCurrentColor(Color colour) {
        this.currentColor = colour;
    }

    public int getShapeCount() {
        return shapes.size();
    }
}

public class DrawingApp extends JFrame {

    private DrawingPanel drawingPanel;
    private JLabel statusLabel;
    private JButton rectangleButton;
    private JButton circleButton;
    private JButton clearButton;
    private JComboBox<String> colorComboBox;

    public DrawingApp() {
        setTitle("Drawing Application");
        setSize(800, 600);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);
        setupUI();
        addActionListeners();
        setVisible(true);
    }

    private void setupUI() {
        JPanel mainPanel = new JPanel(new BorderLayout(10, 10));
        mainPanel.setBorder(BorderFactory.createEmptyBorder(10, 10, 10, 10));

        JPanel topPanel = new JPanel(new FlowLayout(FlowLayout.LEFT));
        JLabel titleLabel = new JLabel("Drawing Application");
        titleLabel.setFont(new Font("Arial", Font.BOLD, 16));
        topPanel.add(titleLabel);

        drawingPanel = new DrawingPanel();
        drawingPanel.setBorder(BorderFactory.createLineBorder(Color.BLACK, 2));

        JPanel controlPanel = new JPanel(new BorderLayout());
        JPanel buttonPanel = new JPanel(new FlowLayout(FlowLayout.LEFT));
        JLabel colorLabel = new JLabel("Colour:");
        colorComboBox = new JComboBox<String>(
                new String[] {"Blue", "Red", "Green", "Yellow", "Orange", "Purple"});
        rectangleButton = new JButton("Add Rectangle");
        circleButton = new JButton("Add Circle");
        clearButton = new JButton("Clear All");

        buttonPanel.add(colorLabel);
        buttonPanel.add(colorComboBox);
        buttonPanel.add(rectangleButton);
        buttonPanel.add(circleButton);
        buttonPanel.add(clearButton);

        statusLabel = new JLabel("Ready. Shapes: 0");
        controlPanel.add(buttonPanel, BorderLayout.CENTER);
        controlPanel.add(statusLabel, BorderLayout.SOUTH);

        mainPanel.add(topPanel, BorderLayout.NORTH);
        mainPanel.add(drawingPanel, BorderLayout.CENTER);
        mainPanel.add(controlPanel, BorderLayout.SOUTH);
        setContentPane(mainPanel);
    }

    private void addActionListeners() {
        rectangleButton.addActionListener(new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent e) {
                drawingPanel.setCurrentColor(getSelectedColor());
                int x = (int) (Math.random() * 600);
                int y = (int) (Math.random() * 350);
                drawingPanel.addRectangle(x, y, 100, 80);
                updateStatus();
            }
        });
        circleButton.addActionListener(new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent e) {
                drawingPanel.setCurrentColor(getSelectedColor());
                int x = 60 + (int) (Math.random() * 600);
                int y = 60 + (int) (Math.random() * 350);
                drawingPanel.addCircle(x, y, 50);
                updateStatus();
            }
        });
        clearButton.addActionListener(new ActionListener() {
            @Override
            public void actionPerformed(ActionEvent e) {
                drawingPanel.clear();
                updateStatus();
            }
        });
    }

    private Color getSelectedColor() {
        String selected = (String) colorComboBox.getSelectedItem();
        if ("Red".equals(selected)) {
            return Color.RED;
        } else if ("Green".equals(selected)) {
            return Color.GREEN;
        } else if ("Yellow".equals(selected)) {
            return Color.YELLOW;
        } else if ("Orange".equals(selected)) {
            return new Color(255, 165, 0);
        } else if ("Purple".equals(selected)) {
            return new Color(128, 0, 128);
        } else {
            return Color.BLUE;
        }
    }

    private void updateStatus() {
        statusLabel.setText("Ready. Shapes: " + drawingPanel.getShapeCount());
    }
}

Discussion

Viva questions (1 of 2)

Why does GamePanel override paintComponent instead of drawing in fire()?
Show the answer
Swing decides when to paint (after resizing, covering, repaint()). Everything drawn must come from paintComponent, or it disappears at the next repaint.
Why is super.paintComponent(g0) the first line?
Show the answer
It clears the panel. Without it, the old picture stays and moving objects leave trails.

Viva questions (2 of 2)

Why are the buttons set to setFocusable(false)?
Show the answer
A clicked button takes the keyboard focus, so the arrow keys would stop reaching the panel.
Why is the window built inside SwingUtilities.invokeLater?
Show the answer
Swing is not thread-safe: all Swing work must happen on the Event Dispatch Thread.
